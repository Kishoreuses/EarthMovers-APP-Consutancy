import 'package:flutter/material.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:firebase_auth/firebase_auth.dart';
import 'package:geolocator/geolocator.dart';
import 'package:geocoding/geocoding.dart';
import 'package:lottie/lottie.dart';
import 'Home.dart';

class Order extends StatefulWidget {
  @override
  _OrderState createState() => _OrderState();
}

class _OrderState extends State<Order> {
  final TextEditingController _addressController = TextEditingController();
  String? _currentLocation;
  bool _isLoading = false;
  bool _isLocating = false;
  final User? user = FirebaseAuth.instance.currentUser;

  List<DocumentSnapshot> cartItems = [];
  Map<String, dynamic>? buyingProduct;
  int buyingQuantity = 1;

  @override
  void didChangeDependencies() {
    super.didChangeDependencies();
    final args = ModalRoute.of(context)?.settings.arguments as Map<String, dynamic>?;

    if (args != null && args.containsKey('product')) {
      // Handle Buy Now logic
      buyingProduct = args['product'].data();
      buyingProduct!['id'] = args['product'].id;
      buyingQuantity = args['quantity'] ?? 1;
      setState(() {});
    } else {
      _fetchCurrentUserCart();
    }
  }

  Future<void> _fetchCurrentUserCart() async {
    setState(() => _isLoading = true);
    var snapshot = await FirebaseFirestore.instance
        .collection('Users')
        .doc(user!.uid)
        .collection('cart')
        .get();
    setState(() {
      cartItems = snapshot.docs;
      _isLoading = false;
    });
  }

  Future<void> _getCurrentLocation() async {
    setState(() => _isLocating = true);
    try {
      bool serviceEnabled = await Geolocator.isLocationServiceEnabled();
      if (!serviceEnabled) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text("Please enable location services")),
        );
        setState(() => _isLocating = false);
        return;
      }

      LocationPermission permission = await Geolocator.checkPermission();
      if (permission == LocationPermission.denied || permission == LocationPermission.deniedForever) {
        permission = await Geolocator.requestPermission();
        if (permission != LocationPermission.whileInUse && permission != LocationPermission.always) {
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text("Location permission denied")),
          );
          setState(() => _isLocating = false);
          return;
        }
      }

      Position position = await Geolocator.getCurrentPosition(desiredAccuracy: LocationAccuracy.high);
      List<Placemark> placemarks = await placemarkFromCoordinates(position.latitude, position.longitude);
      Placemark place = placemarks[0];

      String fullAddress =
          "${place.name}, ${place.street}, ${place.subLocality}, ${place.locality}, ${place.administrativeArea}, ${place.postalCode}";

      setState(() {
        _currentLocation = fullAddress;
        _addressController.text = fullAddress;
      });
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text("Failed to fetch location")));
    }
    setState(() => _isLocating = false);
  }

  Future<void> _placeOrder() async {
    if (_addressController.text.isEmpty && _currentLocation == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text("Please enter address or share location")),
      );
      return;
    }

    String address = _currentLocation ?? _addressController.text;

    if (buyingProduct != null) {
      // Single product from "Buy Now"
      String orderId = FirebaseFirestore.instance.collection('orders').doc().id;
      int totalPrice = buyingProduct!['pricePerUnit'] * buyingQuantity;

      await FirebaseFirestore.instance
          .collection('Users')
          .doc(user!.uid)
          .collection('orders')
          .doc(orderId)
          .set({
        'orderId': orderId,
        'name': buyingProduct!['name'],
        'image_url': buyingProduct!['image_url'],
        'quantity': buyingQuantity,
        'totalPrice': totalPrice,
        'address': address,
        'timestamp': FieldValue.serverTimestamp(),
        'status': 'Pending',
        'driverName': 'Soon',
        'driverPhone': '0000000000',
      });
    } else {
      // Cart-based products
      for (var item in cartItems) {
        String orderId = FirebaseFirestore.instance.collection('orders').doc().id;
        await FirebaseFirestore.instance
            .collection('Users')
            .doc(user!.uid)
            .collection('orders')
            .doc(orderId)
            .set({
          'orderId': orderId,
          'name': item['name'],
          'image_url': item['image_url'],
          'quantity': item['quantity'],
          'totalPrice': item['totalPrice'],
          'address': address,
          'timestamp': FieldValue.serverTimestamp(),
          'status': 'Pending',
          'driverName': 'Soon',
          'driverPhone': '0000000000',
        });
      }

      for (var item in cartItems) {
        await FirebaseFirestore.instance
            .collection('Users')
            .doc(user!.uid)
            .collection('cart')
            .doc(item.id)
            .delete();
      }
    }

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text("Order Placed Successfully!")),
    );

    Navigator.pushAndRemoveUntil(
      context,
      MaterialPageRoute(builder: (context) => Home()),
          (route) => false,
    );
  }

  @override
  Widget build(BuildContext context) {
    final isBuyingNow = buyingProduct != null;

    return Scaffold(
      appBar: AppBar(
        title: Text("Order Summary"),
        backgroundColor: Colors.orangeAccent,
      ),
      body: Stack(
        children: [
          Container(
            decoration: BoxDecoration(
              gradient: LinearGradient(
                colors: [Colors.orange.shade200, Colors.white],
                begin: Alignment.topCenter,
                end: Alignment.bottomCenter,
              ),
            ),
          ),
          _isLoading && !isBuyingNow
              ? Center(child: CircularProgressIndicator())
              : Padding(
            padding: EdgeInsets.all(20),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Lottie.asset("assets/checkout.json", height: 150),
                SizedBox(height: 20),
                Text("Delivery Address", style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
                SizedBox(height: 10),
                TextField(
                  controller: _addressController,
                  decoration: InputDecoration(
                    border: OutlineInputBorder(),
                    hintText: "Enter your address",
                  ),
                ),
                SizedBox(height: 10),
                ElevatedButton.icon(
                  onPressed: _isLocating ? null : _getCurrentLocation,
                  icon: Icon(Icons.location_on),
                  label: _isLocating ? Text("Locating...") : Text("Share Live Location"),
                  style: ElevatedButton.styleFrom(backgroundColor: Colors.blueAccent),
                ),
                if (_currentLocation != null)
                  Padding(
                    padding: const EdgeInsets.only(top: 8.0),
                    child: Text(
                      "Using location: $_currentLocation",
                      style: TextStyle(fontSize: 16, fontWeight: FontWeight.w600),
                    ),
                  ),
                SizedBox(height: 20),
                Divider(),
                Text("Ordered Products", style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
                SizedBox(height: 10),
                Expanded(
                  child: isBuyingNow
                      ? Card(
                    margin: EdgeInsets.only(bottom: 10),
                    shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(10)),
                    child: ListTile(
                      leading: Image.network(buyingProduct!['image_url'], width: 60, fit: BoxFit.cover),
                      title: Text(buyingProduct!['name'], style: TextStyle(fontWeight: FontWeight.bold)),
                      subtitle: Text("Quantity: $buyingQuantity\nPrice: ₹${buyingProduct!['pricePerUnit'] * buyingQuantity}"),
                    ),
                  )
                      : cartItems.isEmpty
                      ? Center(child: Text("No items in cart", style: TextStyle(fontSize: 18)))
                      : ListView.builder(
                    itemCount: cartItems.length,
                    itemBuilder: (context, index) {
                      var item = cartItems[index];
                      return Card(
                        margin: EdgeInsets.only(bottom: 10),
                        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(10)),
                        child: ListTile(
                          leading: Image.network(item['image_url'], width: 60, fit: BoxFit.cover),
                          title: Text(item['name'], style: TextStyle(fontWeight: FontWeight.bold)),
                          subtitle: Text("Quantity: ${item['quantity']}\nPrice: ₹${item['totalPrice']}"),
                        ),
                      );
                    },
                  ),
                ),
                Spacer(),
                ElevatedButton(
                  onPressed: _placeOrder,
                  style: ElevatedButton.styleFrom(
                    backgroundColor: Colors.green,
                    minimumSize: Size(double.infinity, 50),
                  ),
                  child: Text("Proceed to Payment", style: TextStyle(color: Colors.white, fontSize: 18)),
                ),
              ],
            ),
          ),
        ],
      ),
    );
  }
}
